//go:build !wasm

package ui

import (
	"fmt"
	"os/exec"
	"strings"
)

// SelectIFCFile opens the desktop's native file chooser. An empty path means
// that the user cancelled the dialog.
func SelectIFCFile() (string, error) {
	var command *exec.Cmd
	if path, err := exec.LookPath("kdialog"); err == nil {
		command = exec.Command(path, "--getopenfilename", "", "*.ifc|IFC-Dateien", "--title", "STEPView – IFC öffnen")
	} else if path, err := exec.LookPath("zenity"); err == nil {
		command = exec.Command(path, "--file-selection", "--title=STEPView – IFC öffnen", "--file-filter=IFC-Dateien | *.ifc")
	} else {
		return "", fmt.Errorf("kein Dateidialog verfügbar: bitte kdialog oder zenity installieren")
	}
	output, err := command.Output()
	if err != nil {
		if _, cancelled := err.(*exec.ExitError); cancelled {
			return "", nil
		}
		return "", err
	}
	return strings.TrimSpace(string(output)), nil
}
